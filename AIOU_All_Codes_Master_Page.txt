<script>
(function() {
    const searchInput = document.getElementById('mainSearch');
    const resultInfo = document.getElementById('resultInfo');
    const tables = document.querySelectorAll('.modern-table');

    function updateTables() {
        const query = searchInput.value.toLowerCase().trim();
        let foundCount = 0;

        tables.forEach(table => {
            const rows = table.querySelectorAll('tbody tr');
            let hasMatchInTable = false;

            rows.forEach(row => {
                const text = row.innerText.toLowerCase();
                if (text.includes(query)) {
                    row.style.display = '';
                    foundCount++;
                    hasMatchInTable = true;
                } else {
                    row.style.display = 'none';
                }
            });

            const card = table.closest('.table-card');
            const header = card ? card.previousElementSibling : null;
            
            if (query && !hasMatchInTable) {
                if(card) card.style.display = 'none';
                if(header && header.classList.contains('section-header')) header.style.display = 'none';
            } else {
                if(card) card.style.display = '';
                if(header && header.classList.contains('section-header')) header.style.display = '';
            }
        });

        if (query) {
            resultInfo.innerHTML = foundCount > 0 ? `✅ Found ${foundCount}` : `❌ No results`;
        } else {
            resultInfo.innerHTML = '';
        }
    }

    searchInput.addEventListener('input', updateTables);

    // Auto-Search Logic for missing links
    tables.forEach(table => {
        table.querySelectorAll('tr').forEach(row => {
            const codeCell = row.cells[0];
            if (codeCell) {
                const code = codeCell.innerText.trim();
                const buttons = row.querySelectorAll('.dl-btn.missing');
                buttons.forEach(btn => {
                    btn.setAttribute('href', 'https://aioustudio9.com/?s=' + code + '+Spring+2026');
                    btn.setAttribute('target', '_blank');
                    btn.title = "Click to search for this code on Provider site";
                });
            }
        });
    });
})();
</script>
